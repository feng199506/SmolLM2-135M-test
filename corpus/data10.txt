# 先生成基础模板，再扩展为1000条独立问题。
base = []
for cat, items in categories.items():
    for q, a in items:
        base.append((cat, q, a))

samples = []
seen = set()

# 基础样本全部加入
for cat, q, a in base:
    if q not in seen:
        samples.append({"messages":[{"role":"user","content":q},{"role":"assistant","content":a}]})
        seen.add(q)

# 用“同义问法 + 保持答案语义一致”的方式扩充到1000条。
question_prefixes = [
    "请简单解释一下：",
    "请介绍一下：",
    "如何理解",
    "用通俗的话解释",
    "简要说明",
    "请问什么是",
    "能否解释一下",
    "初学者应该如何理解",
    "请给出一个简洁定义：",
    "请概括说明",
]

for cat, q, a in base:
    core = q
    for p in question_prefixes:
        nq = p + core.replace("什么是", "").replace("？", "") + "？"
        if nq not in seen:
            samples.append({"messages":[{"role":"user","content":nq},{"role":"assistant","content":a}]})
            seen.add(nq)
        if len(samples) >= 1000:
            break
    if len(samples) >= 1000:
        break

# 如果仍不足，用带场景的问题扩充，避免简单重复。
scenario_templates = [
    "在实际项目中，为什么需要理解{}？",
    "{}在机器学习项目中有什么作用？",
    "{}与实际工程开发有什么关系？",
    "学习{}时最核心的概念是什么？",
    "{}通常用于解决什么问题？",
    "{}有哪些常见应用场景？",
    "如何用一句话向初学者解释{}？",
    "如果刚开始学习{}，应该先掌握什么？",
    "{}的核心思想是什么？",
    "{}为什么在实际系统中比较常见？",
]
for cat, q, a in base:
    term = q.replace("什么是", "").replace("？", "")
    for t in scenario_templates:
        nq = t.format(term)
        if nq not in seen:
            # 根据原定义生成一个不引入额外事实的回答
            if "为什么需要理解" in nq:
                na = f"理解{term}有助于掌握相关概念和正确使用它。具体价值取决于任务场景，但核心仍是理解其定义、作用和适用范围。"
            elif "有什么作用" in nq or "用于解决什么问题" in nq:
                na = a
            elif "应用场景" in nq:
                na = f"{term}可以用于与其定义相符的相关任务。实际应用时需要结合数据、系统目标和资源条件选择是否使用。"
            else:
                na = a
            samples.append({"messages":[{"role":"user","content":nq},{"role":"assistant","content":na}]})
            seen.add(nq)
        if len(samples) >= 1000:
            break
    if len(samples) >= 1000:
        break

samples = samples[:1000]

out_dir = Path("./data")
jsonl_path = out_dir / "qwen_qa_1000.jsonl"
json_path = out_dir / "qwen_qa_1000.json"

with jsonl_path.open("w", encoding="utf-8") as f:
    for item in samples:
        f.write(json.dumps(item, ensure_ascii=False) + "\n")

with json_path.open("w", encoding="utf-8") as f:
    json.dump(samples, f, ensure_ascii=False, indent=2)
